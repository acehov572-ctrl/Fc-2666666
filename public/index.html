const express = require("express"); const path = require("path"); const { chromium } = require("playwright"); const app = express(); const PORT = process.env.PORT || 10000; const PUBLIC_DIR = path.join(__dirname, "public"); app.use(express.json({ limit: "1mb" })); app.use(express.urlencoded({ extended: true })); // Отдаём сайт из папки public app.use(express.static(PUBLIC_DIR)); app.get("/", (req, res) => { res.sendFile(path.join(PUBLIC_DIR, "index.html")); }); // Проверка сервера app.get("/api/health", (req, res) => { res.json({ ok: true, service: "FC26 Live AI" }); }); // Разбор текста страницы Fonbet function parseMatchText(text) { const lines = String(text || "") .replace(/\r/g, "") .split("\n") .map(s => s.trim()) .filter(Boolean); const idx = lines.findIndex(s => /FC\s*26|Лига\s*Про/i.test(s)); const pool = idx >= 0 ? lines.slice(idx, idx + 100) : lines.slice(0, 150); let home = ""; let away = ""; let homeScore = 0; let awayScore = 0; let minute = 0; for (const line of pool) { let m = line.match( /(.+?)\s*[-–—]\s*(.+?)\s+(\d{1,2})\s*:\s*(\d{1,2})(?:\s+(\d{1,3})\s*(?:'|мин|м))?$/i ); if (m) { home = m[1].trim(); away = m[2].trim(); homeScore = Number(m[3]); awayScore = Number(m[4]); if (m[5]) { minute = Number(m[5]); } break; } m = line.match(/(\d{1,2})\s*:\s*(\d{1,2})/); if (m) { homeScore = Number(m[1]); awayScore = Number(m[2]); } m = line.match(/(\d{1,3})\s*(?:'|мин|минута)\b/i); if (m) { minute = Number(m[1]); } } return { home: home || "Команда 1", away: away || "Команда 2", homeScore, awayScore, minute }; } // Анализ вставленного текста app.post("/api/analyze-text", (req, res) => { try { res.json(parseMatchText(req.body?.text || "")); } catch (e) { res.status(400).json({ error: e.message }); } }); // Анализ ссылки Fonbet app.post("/api/analyze-url", async (req, res) => { const target = String(req.body?.url || "").trim(); if (!target) { return res.status(400).json({ error: "Не указана ссылка на матч." }); } let parsed; try { parsed = new URL(target); } catch { return res.status(400).json({ error: "Неверная ссылка." }); } if (!/(^|\.)fonbet\.(com|ru)$/i.test(parsed.hostname)) { return res.status(400).json({ error: "Разрешены только ссылки Fonbet." }); } let browser; try { browser = await chromium.launch({ headless: true, args: [ "--no-sandbox", "--disable-setuid-sandbox" ] }); const page = await browser.newPage({ viewport: { width: 1280, height: 900 } }); await page.goto(target, { waitUntil: "domcontentloaded", timeout: 30000 }); await page.waitForTimeout(3000); const text = await page.locator("body").innerText({ timeout: 10000 }); res.json({ ...parseMatchText(text), source: "Fonbet", url: target }); } catch (e) { res.status(502).json({ error: "Не удалось получить данные со страницы Fonbet: " + e.message }); } finally { if (browser) { await browser.close().catch(() => {}); } } }); // Для остальных адресов тоже отдаём приложение app.use((req, res) => { res.sendFile(path.join(PUBLIC_DIR, "index.html")); }); app.listen(PORT, "0.0.0.0", () => { console.log( `FC26 Live AI server running on 0.0.0.0:${PORT}` ); });
